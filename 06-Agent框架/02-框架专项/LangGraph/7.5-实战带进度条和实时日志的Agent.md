# 7.5 实战：带进度条和实时日志的 Agent

这一节我们把流式输出的所有知识综合起来，做一个完整的实战项目——一个带进度条和实时日志的 Agent。

---

## 目标

构建一个"研究报告生成 Agent"，它会：
1. 接收用户的研究主题
2. 分步骤执行（搜索、分析、生成报告）
3. 实时显示每个步骤的进度
4. 流式输出 LLM 生成的内容

---

## 完整代码

```python
from dotenv import load_dotenv
from langgraph.graph import StateGraph, MessagesState, START, END
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.messages import ToolMessage
import time

load_dotenv()

# ========== 工具 ==========
@tool
def search_research(topic: str) -> str:
    """搜索研究资料"""
    time.sleep(2)  # 模拟耗时搜索
    return f"关于《{topic}》的研究资料：\n1. 最新进展...\n2. 核心观点...\n3. 未来趋势..."

@tool
def analyze_data(data: str) -> str:
    """分析数据"""
    time.sleep(1)  # 模拟分析
    return "分析结果：数据显示该领域正在快速发展..."

# ========== LLM ==========
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0, streaming=True)
tools = [search_research, analyze_data]
llm_with_tools = llm.bind_tools(tools)
tool_map = {tool.name: tool for tool in tools}

# ========== 节点 ==========
def search_node(state):
    """搜索阶段"""
    print("\n" + "="*60)
    print("📚 阶段 1/3：搜索研究资料")
    print("="*60)
    
    topic = state["messages"][-1].content
    result = search_research.invoke(topic)
    
    print(f"✅ 搜索完成")
    return {"messages": [ToolMessage(content=result, tool_call_id="search_1")]}

def analyze_node(state):
    """分析阶段"""
    print("\n" + "="*60)
    print("📊 阶段 2/3：分析数据")
    print("="*60)
    
    last_tool_result = state["messages"][-1].content
    result = analyze_data.invoke(last_tool_result)
    
    print(f"✅ 分析完成")
    return {"messages": [ToolMessage(content=result, tool_call_id="analyze_1")]}

def generate_report_node(state):
    """生成报告阶段"""
    print("\n" + "="*60)
    print("📝 阶段 3/3：生成研究报告")
    print("="*60)
    
    response = llm_with_tools.invoke(state["messages"])
    return {"messages": [response]}

# ========== State ==========
from typing import TypedDict
from langchain_core.messages import BaseMessage

class ReportState(TypedDict):
    messages: list[BaseMessage]
    phase: str

# ========== 构建图 ==========
graph = StateGraph(ReportState)
graph.add_node("search", search_node)
graph.add_node("analyze", analyze_node)
graph.add_node("generate", generate_report_node)

graph.add_edge(START, "search")
graph.add_edge("search", "analyze")
graph.add_edge("analyze", "generate")
graph.add_edge("generate", END)

app = graph.compile()

# ========== 流式执行 ==========
def run_with_streaming(topic: str):
    """带流式输出的执行"""
    
    input_data = {"messages": [{"role": "user", "content": f"研究主题：{topic}"}]}
    
    for event in app.stream_events(input_data, version="v2"):
        event_type = event["event"]
        
        # 阶段提示
        if event_type == "on_chain_start":
            node_name = event["name"]
            if node_name == "search":
                print_progress("搜索阶段", 1, 3)
            elif node_name == "analyze":
                print_progress("分析阶段", 2, 3)
            elif node_name == "generate":
                print_progress("生成报告", 3, 3)
        
        # LLM token（打字机效果）
        elif event_type == "on_chat_model_stream":
            content = event["data"]["chunk"].content
            if content:
                print(content, end="", flush=True)
        
        # 工具调用
        elif event_type == "on_tool_start":
            print(f"\n  🔧 调用工具：{event['name']}")
        
        # 工具完成
        elif event_type == "on_tool_end":
            print(f"  ✅ 工具完成")
        
        # 阶段完成
        elif event_type == "on_chain_end":
            node_name = event["name"]
            if node_name in ["search", "analyze", "generate"]:
                print(f"\n✅ {node_name} 阶段完成\n")

def print_progress(phase: str, current: int, total: int):
    """打印进度条"""
    percentage = int(current / total * 100)
    bar_length = 30
    filled = int(bar_length * current / total)
    bar = "█" * filled + "░" * (bar_length - filled)
    
    print(f"\n{'─'*60}")
    print(f"  {phase}")
    print(f"  [{bar}] {percentage}%")
    print(f"{'─'*60}\n")

# ========== 测试 ==========
if __name__ == "__main__":
    print("\n🚀 开始生成研究报告\n")
    run_with_streaming("人工智能在教育领域的应用")
    print("\n\n✅ 研究报告生成完成！")
```

## 小结

通过这个实战项目，我们综合运用了：

1. **stream_events()**：获取细粒度的事件流
2. **on_chat_model_stream**：实时显示 LLM 输出
3. **on_tool_start / on_tool_end**：显示工具调用
4. **on_chain_start / on_chain_end**：显示节点执行
5. **进度条**：可视化执行进度

掌握了流式输出，你就能构建用户体验更好的 Agent 应用了！
