# 3.6 条件路由（Conditional Edges）深入

条件路由是 LangGraph 最强大的特性之一——它让 Agent 能够"自己决定下一步做什么"。

这一节我们深入学习 `add_conditional_edges()` 的各种用法和技巧。

---

## 回顾：add_conditional_edges() 的语法

```python
graph.add_conditional_edges(
    source: str,           # 从哪个节点出来
    path: Callable,        # 路由函数
    path_map: dict         # 路径映射
)
```

执行流程：
1. 执行完 `source` 节点
2. 调用 `path` 函数，传入当前 State
3. `path` 函数返回一个字符串（比如 `"branch_a"`）
4. 在 `path_map` 里查找这个字符串，得到下一个节点
5. 跳转到那个节点

---

## 路由函数的编写技巧

### 技巧 1：基于 State 的简单判断

```python
def router(state):
    if state["count"] > 10:
        return "high"
    else:
        return "low"

graph.add_conditional_edges(
    "check",
    router,
    {
        "high": "process_high",
        "low": "process_low"
    }
)
```

### 技巧 2：多条件判断

```python
def router(state):
    if state["error"]:
        return "error"
    elif state["count"] > 100:
        return "done"
    elif state["need_retry"]:
        return "retry"
    else:
        return "continue"

graph.add_conditional_edges(
    "main",
    router,
    {
        "error": "error_handler",
        "done": END,
        "retry": "main",  # 循环回自己
        "continue": "next_step"
    }
)
```

### 技巧 3：基于 LLM 输出的路由

```python
def router(state):
    last_message = state["messages"][-1]
    
    # 如果 LLM 返回了工具调用
    if hasattr(last_message, "tool_calls") and last_message.tool_calls:
        return "tools"
    
    # 如果 LLM 说"需要更多信息"
    if "需要更多信息" in last_message.content:
        return "clarify"
    
    # 否则就结束
    return "end"

graph.add_conditional_edges(
    "llm",
    router,
    {
        "tools": "call_tools",
        "clarify": "ask_user",
        "end": END
    }
)
```

### 技巧 4：用枚举增强可读性

```python
from enum import Enum

class Route(str, Enum):
    TOOLS = "tools"
    CLARIFY = "clarify"
    END = "end"

def router(state):
    last_message = state["messages"][-1]
    
    if hasattr(last_message, "tool_calls") and last_message.tool_calls:
        return Route.TOOLS
    elif "需要更多信息" in last_message.content:
        return Route.CLARIFY
    else:
        return Route.END

graph.add_conditional_edges(
    "llm",
    router,
    {
        Route.TOOLS: "call_tools",
        Route.CLARIFY: "ask_user",
        Route.END: END
    }
)
```

## 常见路由模式

### 模式 1：循环 vs 结束

最经典的 Agent 模式：反复调用工具，直到不再需要。

```python
def should_continue(state):
    last_message = state["messages"][-1]
    if hasattr(last_message, "tool_calls") and last_message.tool_calls:
        return "continue"
    else:
        return "end"

graph.add_conditional_edges(
    "llm",
    should_continue,
    {
        "continue": "tools",
        "end": END
    }
)
graph.add_edge("tools", "llm")  # 工具 → LLM，形成循环
```

### 模式 2：错误重试

出错时回到上一步重试，超过次数就放弃。

```python
def retry_or_fail(state):
    if state["error"] and state["retry_count"] < 3:
        return "retry"
    elif state["error"]:
        return "fail"
    else:
        return "success"

graph.add_conditional_edges(
    "risky_node",
    retry_or_fail,
    {
        "retry": "risky_node",  # 回到自己
        "fail": "error_handler",
        "success": "next_step"
    }
)
```

### 模式 3：多工具分发

根据 LLM 想调用的工具，路由到不同的节点。

```python
def route_tools(state):
    last_message = state["messages"][-1]
    if not last_message.tool_calls:
        return "end"
    
    # 取第一个工具调用的名字
    tool_name = last_message.tool_calls[0]["name"]
    return tool_name

graph.add_conditional_edges(
    "llm",
    route_tools,
    {
        "search": "search_tool",
        "calculator": "calculator_tool",
        "database": "database_tool",
        "end": END
    }
)
```

### 模式 4：用户确认

需要人工确认时暂停，确认后继续。

```python
def need_approval(state):
    if state["requires_approval"]:
        return "wait_approval"
    else:
        return "execute"

graph.add_conditional_edges(
    "plan",
    need_approval,
    {
        "wait_approval": "human_approval",
        "execute": "execute_plan"
    }
)
```

## 高级用法：path_map 的灵活性

### 用法 1：多个路由到同一节点

```python
graph.add_conditional_edges(
    "decision",
    router,
    {
        "option_a": "handler",
        "option_b": "handler",  # 和 option_a 一样
        "option_c": "other_handler"
    }
)
```

### 用法 2：路由到 END

```python
graph.add_conditional_edges(
    "check",
    router,
    {
        "done": END,
        "continue": "next_node"
    }
)
```

### 用法 3：动态路由（路由函数返回节点名）

有时候 path_map 是"一对一映射"，可以简化：

```python
def router(state):
    # 直接返回节点名
    return state["next_node"]

graph.add_conditional_edges(
    "dispatcher",
    router,
    {
        "node_a": "node_a",
        "node_b": "node_b",
        "node_c": "node_c"
    }
)
```

**更简洁的写法**（如果所有可能的路由都是节点名）：

```python
# 某些 LangGraph 版本支持
graph.add_conditional_edges(
    "dispatcher",
    lambda state: state["next_node"]  # 直接返回节点名
)
```

## 实战示例：ReAct Agent 的路由逻辑

ReAct（Reasoning + Acting）是一种经典的 Agent 模式：
1. LLM 思考（Reasoning）
2. 决定是否需要工具（Acting）
3. 如果需要，调用工具，然后回到步骤 1
4. 如果不需要，输出最终答案

```python
from langgraph.graph import StateGraph, MessagesState, START, END
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool

llm = ChatOpenAI(model="gpt-4o-mini")

@tool
def search(query: str) -> str:
    """搜索工具"""
    return f"搜索结果：{query}"

tools = [search]
llm_with_tools = llm.bind_tools(tools)

def call_llm(state: MessagesState):
    response = llm_with_tools.invoke(state["messages"])
    return {"messages": [response]}

def call_tools(state: MessagesState):
    last_message = state["messages"][-1]
    tool_calls = last_message.tool_calls
    
    results = []
    for tool_call in tool_calls:
        tool_name = tool_call["name"]
        tool_args = tool_call["args"]
        
        # 调用工具
        if tool_name == "search":
            result = search.invoke(tool_args)
        
        # 构造工具返回消息
        results.append({
            "role": "tool",
            "content": result,
            "tool_call_id": tool_call["id"]
        })
    
    return {"messages": results}

def should_continue(state: MessagesState):
    last_message = state["messages"][-1]
    if hasattr(last_message, "tool_calls") and last_message.tool_calls:
        return "tools"
    else:
        return "end"

# 构建图
graph = StateGraph(MessagesState)
graph.add_node("llm", call_llm)
graph.add_node("tools", call_tools)

graph.add_edge(START, "llm")
graph.add_conditional_edges(
    "llm",
    should_continue,
    {
        "tools": "tools",
        "end": END
    }
)
graph.add_edge("tools", "llm")

app = graph.compile()

# 测试
result = app.invoke({
    "messages": [{"role": "user", "content": "搜索：Python 教程"}]
})

for msg in result["messages"]:
    print(f"{msg.role}: {msg.content}")
```

## 调试技巧

### 技巧 1：在路由函数里打印日志

```python
def router(state):
    decision = "tools" if state["need_tools"] else "end"
    print(f"路由决策：{decision}")
    return decision
```

### 技巧 2：验证所有路径都在 path_map 里

```python
def router(state):
    possible_paths = ["path_a", "path_b", "path_c"]
    decision = state["decision"]
    
    if decision not in possible_paths:
        raise ValueError(f"未知路径：{decision}")
    
    return decision
```

### 技巧 3：用 stream() 查看执行路径

```python
for event in app.stream(initial_state):
    print(f"执行节点：{list(event.keys())[0]}")
```

## 常见错误

### 错误 1：路由函数返回值不在 path_map 里

```python
def router(state):
    return "unknown_path"  # 这个值不在 path_map 里

graph.add_conditional_edges(
    "node",
    router,
    {
        "path_a": "node_a",
        "path_b": "node_b"
    }
)

app = graph.compile()
app.invoke(...)  # 运行时报错：Path 'unknown_path' not found
```

**解决**：确保路由函数的所有可能返回值，都在 path_map 里。

### 错误 2：路由函数没有返回值

```python
def router(state):
    if state["value"] > 10:
        return "high"
    # 如果 value <= 10，没有返回值！

graph.add_conditional_edges(
    "node",
    router,
    {
        "high": "node_high",
        "low": "node_low"
    }
)
```

**运行时报错**：路由函数必须返回字符串。

**解决**：确保所有分支都有 return。

### 错误 3：路由函数抛出异常

```python
def router(state):
    return state["decision"]  # 如果 state 里没有 "decision"，会 KeyError

graph.add_conditional_edges(...)
app.invoke({"value": 10})  # 报错！
```

**解决**：用 `state.get("decision", "default")` 提供默认值。

---

## 小结

条件路由是 LangGraph 的核心能力，让 Agent 能够根据状态动态决策：

- **路由函数**：接收 State，返回字符串
- **path_map**：把返回值映射到节点名
- **常见模式**：
  - 循环 vs 结束
  - 错误重试
  - 多工具分发
  - 用户确认
  
**设计建议**：
1. 路由函数要简单清晰，一眼看懂决策逻辑
2. 用枚举增强可读性
3. 确保所有可能的返回值都在 path_map 里
4. 调试时打印路由决策

掌握了条件路由，你就能构建真正"智能"的 Agent——它会根据情况做出不同的选择。
