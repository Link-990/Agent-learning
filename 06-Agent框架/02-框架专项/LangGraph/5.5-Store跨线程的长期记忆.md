# 5.5 Store：跨线程的长期记忆

Checkpointer 解决了"对话历史"的问题，但还有一类需求它解决不了：**跨对话的长期记忆**。

这就是 **Store 系统**的作用。

---

## Store 是什么?

**Store** 是 LangGraph 提供的**跨线程的键值存储系统**。

与 Checkpointer 的区别:

| 特性 | Checkpointer | Store |
|------|-------------|-------|
| 保存内容 | 完整的执行状态 | 结构化的键值对 |
| 作用域 | 每个 thread_id 独立 | 跨 thread 共享 |
| 自动保存 | ✅ 每个节点后自动 | ❌ 需要手动调用 |
| 查询方式 | 按 thread_id | 按 namespace + 搜索条件 |

---

## Store 的核心概念

### 1. Namespace（命名空间）

用元组表示的层级结构:

```python
("user", "user_123")              # 用户 user_123 的数据
("user", "user_123", "preferences")  # 用户 user_123 的偏好设置
("faq",)                          # FAQ 知识库
("product", "prod_456")           # 产品 prod_456 的信息
```

### 2. Key（键）

在 namespace 下的唯一标识:

```python
namespace = ("user", "user_123")
key = "language"
value = {"lang": "zh-CN"}
```

### 3. Item（条目）

Store 里的一条数据:

```python
{
    "namespace": ("user", "user_123"),
    "key": "language",
    "value": {"lang": "zh-CN"},
    "created_at": "2024-01-15T10:00:00Z",
    "updated_at": "2024-01-15T10:00:00Z"
}
```

---

## Store 的基本操作

### 1. put（保存数据）

```python
store.put(
    namespace=("user", "user_123"),
    key="language",
    value={"lang": "zh-CN"}
)
```

### 2. get（获取单条数据）

```python
item = store.get(
    namespace=("user", "user_123"),
    key="language"
)
print(item.value)  # {"lang": "zh-CN"}
```

### 3. search（搜索数据）

```python
# 获取某个 namespace 下的所有数据
items = store.search(namespace=("user", "user_123"))

for item in items:
    print(f"{item.key}: {item.value}")
```

### 4. delete（删除数据）

```python
store.delete(
    namespace=("user", "user_123"),
    key="language"
)
```

---

## InMemoryStore：内存中的 Store

最简单的 Store 实现——数据保存在内存中。

### 创建

```python
from langgraph.store.memory import InMemoryStore

store = InMemoryStore()
```

### 完整示例

```python
from langgraph.store.memory import InMemoryStore

store = InMemoryStore()

# 保存用户偏好
store.put(("user", "alice"), "language", {"lang": "zh-CN"})
store.put(("user", "alice"), "theme", {"theme": "dark"})
store.put(("user", "bob"), "language", {"lang": "en-US"})

# 获取单条数据
item = store.get(("user", "alice"), "language")
print(item.value)  # {"lang": "zh-CN"}

# 搜索某个用户的所有偏好
items = store.search(("user", "alice"))
print(f"Alice 有 {len(items)} 条偏好设置:")
for item in items:
    print(f"  {item.key}: {item.value}")

# 删除数据
store.delete(("user", "alice"), "theme")

# 再次搜索
items = store.search(("user", "alice"))
print(f"删除后，Alice 有 {len(items)} 条偏好设置")
```

输出:

```
{"lang": "zh-CN"}
Alice 有 2 条偏好设置:
  language: {"lang": "zh-CN"}
  theme: {"theme": "dark"}
删除后，Alice 有 1 条偏好设置
```

## 在 Agent 中使用 Store

### 方式 1：在 @entrypoint 中配置

```python
from langgraph.func import entrypoint, task
from langgraph.store.memory import InMemoryStore

store = InMemoryStore()

@task
def save_preference(user_id: str, key: str, value: dict, store):
    """保存用户偏好"""
    store.put(("user", user_id), key, value)
    return f"已保存 {key}"

@task
def load_preference(user_id: str, key: str, store):
    """读取用户偏好"""
    item = store.get(("user", user_id), key)
    if item:
        return item.value
    return None

@entrypoint(store=store)
def workflow(user_id: str, action: str, key: str = None, value: dict = None):
    if action == "save":
        return save_preference(user_id, key, value)
    elif action == "load":
        result = load_preference(user_id, key)
        return result or "未找到"

# 测试
workflow("alice", "save", "language", {"lang": "zh-CN"})
result = workflow("alice", "load", "language")
print(result)  # {"lang": "zh-CN"}
```

### 方式 2：在 StateGraph 中使用

```python
from langgraph.graph import StateGraph, START, END
from langgraph.store.memory import InMemoryStore
from typing import TypedDict

store = InMemoryStore()

class State(TypedDict):
    user_id: str
    message: str

def load_preferences(state: State):
    """从 Store 加载用户偏好"""
    user_id = state["user_id"]
    items = store.search(("user", user_id))
    
    prefs = {item.key: item.value for item in items}
    return {"message": f"用户偏好: {prefs}"}

graph = StateGraph(State)
graph.add_node("load_prefs", load_preferences)
graph.add_edge(START, "load_prefs")
graph.add_edge("load_prefs", END)

app = graph.compile()

# 先保存一些偏好
store.put(("user", "alice"), "language", {"lang": "zh-CN"})
store.put(("user", "alice"), "theme", {"theme": "dark"})

# 调用 Agent
result = app.invoke({"user_id": "alice"})
print(result["message"])
```

## Store 的高级功能

### 1. 过滤搜索

```python
# 搜索所有用户的语言偏好
items = store.search(
    namespace_prefix=("user",),
    filter={"key": "language"}
)
```

### 2. 分页

```python
# 获取前 10 条
items = store.search(("user", "alice"), limit=10)

# 获取第 11-20 条
items = store.search(("user", "alice"), limit=10, offset=10)
```

### 3. 语义搜索（需要支持的 Store 实现）

```python
# 使用嵌入向量搜索相似内容
items = store.search(
    namespace=("documents",),
    query="如何退款?",
    limit=5
)
```

## Store 的使用场景

### 场景 1：用户偏好设置

```python
# 保存用户的语言、主题、通知偏好
store.put(("user", user_id), "language", {"lang": "zh-CN"})
store.put(("user", user_id), "theme", {"theme": "dark"})
store.put(("user", user_id), "notifications", {"email": True, "sms": False})

# 在任何对话中读取
@task
def personalized_greeting(user_id: str, store):
    lang_item = store.get(("user", user_id), "language")
    lang = lang_item.value.get("lang", "en-US") if lang_item else "en-US"
    
    if lang == "zh-CN":
        return "你好！"
    else:
        return "Hello!"
```

### 场景 2：知识库（FAQ）

```python
# 保存常见问题
store.put(("faq",), "how_to_refund", {
    "question": "如何退款？",
    "answer": "请联系客服，提供订单号即可申请退款。"
})

store.put(("faq",), "shipping_time", {
    "question": "发货需要多久？",
    "answer": "一般 3-5 个工作日发货。"
})

# 在 Agent 中查询
@task
def answer_faq(question: str, store):
    items = store.search(("faq",))
    
    for item in items:
        if question in item.value.get("question", ""):
            return item.value.get("answer")
    
    return "抱歉，没有找到相关问题。"
```

### 场景 3：跨对话的记忆

```python
# 在对话 1 中记录用户信息
store.put(("user", "alice", "memory"), "city", {"city": "北京"})

# 在对话 2（新的 thread_id）中依然能读取
@entrypoint(store=store)
def workflow(user_id: str, store):
    city_item = store.get(("user", user_id, "memory"), "city")
    if city_item:
        city = city_item.value.get("city")
        return f"你住在 {city}"
    return "不知道你住在哪里"

# 无论 thread_id 是什么，都能读取到 "北京"
result = workflow("alice")
print(result)  # "你住在 北京"
```

## Checkpointer + Store：完整的记忆系统

实际应用中，通常同时使用两者:

```python
from langgraph.func import entrypoint, task
from langgraph.checkpoint.memory import MemorySaver
from langgraph.store.memory import InMemoryStore

checkpointer = MemorySaver()  # 对话历史
store = InMemoryStore()       # 长期记忆

@task
def greet_user(user_id: str, store):
    """从 Store 读取用户偏好"""
    lang_item = store.get(("user", user_id), "language")
    lang = lang_item.value.get("lang", "en-US") if lang_item else "en-US"
    
    if lang == "zh-CN":
        return "你好！有什么可以帮你？"
    else:
        return "Hello! How can I help you?"

@task
def chat(user_input: str, previous: str):
    """对话（这里简化，实际应调用 LLM）"""
    return f"回复: {user_input}"

@entrypoint(checkpointer=checkpointer, store=store)
def chat_agent(user_id: str, user_input: str):
    # 1. 从 Store 读取偏好，个性化问候
    greeting = greet_user(user_id)
    
    # 2. 对话（Checkpointer 自动记住历史）
    response = chat(user_input)
    
    return f"{greeting}\n{response}"

# 保存用户偏好
store.put(("user", "alice"), "language", {"lang": "zh-CN"})

# 第一轮对话
result = chat_agent("alice", "你好", config={"thread_id": "1"})
print(result)

# 第二轮对话（同一个 thread，Checkpointer 记住历史）
result = chat_agent("alice", "天气怎么样", config={"thread_id": "1"})
print(result)

# 第三轮对话（新 thread，但 Store 依然记得偏好）
result = chat_agent("alice", "你好", config={"thread_id": "2"})
print(result)
```

## 小结

Store 系统的特点:

**核心概念**:
- **Namespace**: 层级化的命名空间
- **Key**: 在 namespace 下的唯一标识
- **Item**: 一条数据（namespace + key + value）

**基本操作**:
- `put()`: 保存数据
- `get()`: 获取单条数据
- `search()`: 搜索数据
- `delete()`: 删除数据

**使用场景**:
- 用户偏好设置
- 知识库（FAQ）
- 跨对话的记忆

**与 Checkpointer 的配合**:
- Checkpointer: 对话历史（每个 thread 独立）
- Store: 长期记忆（跨 thread 共享）
