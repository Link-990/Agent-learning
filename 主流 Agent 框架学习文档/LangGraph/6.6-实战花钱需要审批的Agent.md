# 6.6 实战：构建一个"花钱需要审批"的 Agent

前面学了这么多理论，现在我们来做一个完整的实战项目——构建一个"花钱需要审批"的 Agent。

这个 Agent 能够：
- 理解用户的消费请求
- 生成消费计划
- 暂停等待人工审批
- 根据审批结果执行或取消

---

## 需求分析

### 场景

你是一个公司的财务助手，员工可以向你提出消费请求，但所有超过 1000 元的消费需要经理审批。

### 功能

1. **消费请求**：用户提出消费请求（描述、金额）
2. **自动判断**：如果金额 > 1000，需要审批；否则直接执行
3. **审批流程**：暂停，等待经理审批（批准、修改、拒绝）
4. **执行消费**：根据审批结果执行

---

## 实现步骤

### 第一步：定义工具

```python
from langchain_core.tools import tool

def execute_payment(amount: float, description: str) -> str:

def check_budget() -> str:
    """查询预算"""
    # 模拟预算数据
    return "当前部门预算：100,000 元，已使用：45,000 元，剩余：55,000 元"
```

### 第二步：定义节点

```python
from langchain_openai import ChatOpenAI
from langchain_core.messages import SystemMessage

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

def parse_request_node(state):
    """解析用户的消费请求"""
    user_input = state["messages"][-1].content
    
    # 用 LLM 提取金额和描述
    response = llm.invoke([
        SystemMessage(content="""
从用户输入中提取消费请求，返回 JSON 格式：
{"amount": 数字, "description": "描述"}
"""),
        {"role": "user", "content": user_input}
    ])
    
    # 简化版：直接解析（实际应该用 JSON 解析）
    # 这里假设 response 已经包含了金额和描述
    amount = 1500  # 实际从 response 提取
    description = "购买办公用品"
    
    return {"amount": amount, "description": description}

def check_threshold_node(state):
    """检查是否需要审批"""
    amount = state["amount"]
    
    if amount > 1000:
        return {"need_approval": True, "status": "pending_approval"}
    else:
        return {"need_approval": False, "status": "approved"}

def generate_plan_node(state):
    """生成消费计划"""
    amount = state["amount"]
    description = state["description"]
    
    plan = f"消费计划：{description}，金额 {amount} 元"
    return {"plan": plan, "messages": [{"role": "assistant", "content": plan}]}

def execute_node(state):
    """执行消费"""
    amount = state["amount"]
    description = state["description"]
    
    result = execute_payment.invoke({"amount": amount, "description": description})
    return {"messages": [{"role": "assistant", "content": result}]}

def reject_node(state):
    """拒绝消费"""
    return {"messages": [{"role": "assistant", "content": "❌ 消费请求已被拒绝"}]}
```

### 第三步：构建图

```python
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import MemorySaver
from typing import TypedDict

class State(TypedDict):
    messages: list
    amount: float
    description: str
    need_approval: bool
    status: str
    plan: str

graph = StateGraph(State)

graph.add_node("parse", parse_request_node)
graph.add_node("check", check_threshold_node)
graph.add_node("plan", generate_plan_node)
graph.add_node("execute", execute_node)
graph.add_node("reject", reject_node)

# 流程
graph.add_edge(START, "parse")
graph.add_edge("parse", "check")

# 条件路由：是否需要审批？
def route_by_approval(state):
    if state.get("need_approval"):
        return "plan"  # 需要审批，先生成计划
    else:
        return "execute"  # 不需要审批，直接执行

graph.add_conditional_edges("check", route_by_approval, {
    "plan": "plan",
    "execute": "execute"
})

# 需要审批的，在 plan 之后暂停
graph.add_edge("plan", END)  # 这里会暂停
graph.add_edge("execute", END)
graph.add_edge("reject", END)

# 在 plan 节点后暂停（等待审批）
checkpointer = MemorySaver()
app = graph.compile(
    checkpointer=checkpointer,
    interrupt_after=["plan"]  # plan 执行完后暂停
)
```

### 第四步：测试

```python
from langgraph.types import Command

# 测试 1：金额小于 1000，直接执行
thread_id_1 = "test_1"
result = app.invoke(
    {"messages": [{"role": "user", "content": "购买文具，500 元"}]},
    config={"configurable": {"thread_id": thread_id_1}}
)
print(f"测试 1：{result['messages'][-1]['content']}")
# 输出：✅ 已支付 500 元，用途：购买文具

# 测试 2：金额大于 1000，需要审批
thread_id_2 = "test_2"
result = app.invoke(
    {"messages": [{"role": "user", "content": "购买设备，1500 元"}]},
    config={"configurable": {"thread_id": thread_id_2}}
)

# 查看状态：应该在 plan 之后暂停
state = app.get_state(config={"configurable": {"thread_id": thread_id_2}})
print(f"消费计划：{state.values['plan']}")
print(f"状态：{state.values['status']}")  # pending_approval

# 经理审批：批准
result = app.invoke(
    Command(resume={"approved": True}),
    config={"configurable": {"thread_id": thread_id_2}}
)
print(f"测试 2：{result['messages'][-1]['content']}")
# 输出：✅ 已支付 1500 元，用途：购买设备

# 测试 3：经理拒绝
thread_id_3 = "test_3"
app.invoke(
    {"messages": [{"role": "user", "content": "团建活动，2000 元"}]},
    config={"configurable": {"thread_id": thread_id_3}}
)

# 经理拒绝
result = app.invoke(
    Command(resume={"approved": False}),
    config={"configurable": {"thread_id": thread_id_3}}
)
print(f"测试 3：{result['messages'][-1]['content']}")
# 输出：❌ 消费请求已被拒绝
```

## 扩展方向

这个 Agent 还可以扩展：

### 1. 多级审批

金额超过 10000 元需要总监审批，超过 50000 元需要 CEO 审批。

### 2. 审批历史记录

把每次审批的结果保存到 Store，方便审计。

### 3. 通知系统

审批请求通过邮件、短信、推送等方式通知审批人。

### 4. 审批超时

如果 24 小时内没有审批，自动提醒或升级。

### 5. 预算检查

在执行前检查部门预算是否充足。

## 小结

通过这个实战项目，我们综合运用了：

1. **Checkpointer**：保存审批状态
2. **interrupt**：暂停等待审批
3. **Command(resume=...)**：恢复执行
4. **条件路由**：根据金额决定是否需要审批

这个 Agent 具备了：
- ✅ 自动判断是否需要审批
- ✅ 暂停等待人工审批
- ✅ 根据审批结果执行或取消

掌握了这些技能，你就能构建各种需要人工介入的 Agent 应用了！
