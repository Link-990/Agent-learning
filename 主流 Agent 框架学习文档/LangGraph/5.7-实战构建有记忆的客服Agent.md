# 5.7 实战：构建一个有记忆的客服 Agent

前面我们学了 Checkpointer（对话历史）和 Store（长期记忆），现在把它们综合起来，构建一个**真正有记忆的客服 Agent**。

---

## 需求分析

我们要构建的客服 Agent 需要：

### 1. 对话记忆（Checkpointer）
- 记住当前对话的内容
- 支持多轮对话
- 不同用户之间隔离

### 2. 长期记忆（Store）
- 记住用户的基本信息（姓名、偏好）
- 保存常见问题答案（FAQ）
- 跨对话的记忆（比如用户上次提过的问题）

### 3. 工具能力
- 搜索知识库（FAQ）
- 获取用户偏好
- 调用外部工具（订单查询、物流查询）

---

## 架构设计

```
用户消息
    ↓
[加载用户记忆] ← Store
    ↓
[FAQ 匹配] ← Store
    ↓
[LLM 处理] ← 如果有匹配，结合 FAQ 回答
    ↓
[调用工具] ← 如果需要查询订单、物流等
    ↓
[保存对话] ← Checkpointer
    ↓
返回回复
```

## 实现步骤

### 第一步：定义工具

```python
from langchain_core.tools import tool

@tool
def search_faq(question: str) -> str:
    """搜索常见问题知识库"""
    # 这里模拟，实际应该从 Store 或数据库读取
    faqs = {
        "退款": "申请退款请联系客服，提供订单号。一般 3-5 个工作日处理。",
        "物流": "发货后一般 3-5 个工作日送达，可以在订单详情里查看物流信息。",
        "优惠券": "优惠券在个人中心-我的优惠券里查看和使用。",
        "售后": "如有售后问题，请提供订单号和问题描述，我们会及时为您处理。"
    }
    
    for key, answer in faqs.items():
        if key in question:
            return answer
    
    return "抱歉，没有找到相关问题。请联系人工客服。"

@tool
def query_order(order_id: str) -> str:
    """查询订单状态"""
    # 模拟订单数据
    orders = {
        "1001": {"status": "已发货", "product": "iPhone 15"},
        "1002": {"status": "待付款", "product": "MacBook Pro"},
    }
    
    if order_id in orders:
        order = orders[order_id]
        return f"订单 {order_id}（{order['product']}）状态：{order['status']}"
    else:
        return f"未找到订单 {order_id}"

@tool
def get_user_preference(user_id: str) -> str:
    """获取用户偏好"""
    # 模拟用户数据
    preferences = {
        "alice": {"language": "中文", "theme": "深色模式"},
        "bob": {"language": "English", "theme": "light"},
    }
    
    if user_id in preferences:
        prefs = preferences[user_id]
        return f"用户偏好：语言={prefs['language']}，主题={prefs['theme']}"
    else:
        return "未找到用户偏好"
```

### 第二步：定义节点

```python
from langchain_openai import ChatOpenAI
from langchain_core.messages import SystemMessage

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

def greet_node(state):
    """问候节点：从 Store 加载用户偏好"""
    user_id = state["user_id"]
    
    # 从 Store 读取用户偏好
    prefs = store.search(("user", user_id, "preferences"))
    
    if prefs:
        user_info = {item.key: item.value for item in prefs}
        greeting = f"欢迎回来！根据您的偏好，使用 {user_info.get('language', '中文')} 为您服务。"
    else:
        greeting = "欢迎！请问有什么可以帮您？"
    
    return {"messages": [SystemMessage(content=greeting)]}

def faq_match_node(state):
    """FAQ 匹配节点"""
    user_message = state["messages"][-1].content
    faq_answer = search_faq(user_message)
    
    if "抱歉" not in faq_answer:
        return {"faq_answer": faq_answer}
    return {}

def llm_node(state):
    """LLM 处理节点"""
    messages = state["messages"]
    
    # 如果有 FAQ 匹配，注入到系统提示
    if state.get("faq_answer"):
        system_prompt = f"以下是相关 FAQ 信息：\n{state['faq_answer']}\n\n请结合以上信息回答用户的问题。"
        messages = [SystemMessage(content=system_prompt)] + messages
    
    response = llm.invoke(messages)
    return {"messages": [response]}

def tool_node(state):
    """工具调用节点"""
    last_message = state["messages"][-1]
    
    if not hasattr(last_message, "tool_calls") or not last_message.tool_calls:
        return {}
    
    tool_map = {
        "search_faq": search_faq,
        "query_order": query_order,
        "get_user_preference": get_user_preference
    }
    
    results = []
    for tool_call in last_message.tool_calls:
        tool_name = tool_call["name"]
        tool_args = tool_call["args"]
        
        tool_func = tool_map[tool_name]
        result = tool_func.invoke(tool_args)
        
        results.append({
            "role": "tool",
            "content": result,
            "tool_call_id": tool_call["id"]
        })
    
    return {"messages": results}

def save_memory_node(state):
    """保存用户记忆到 Store"""
    user_id = state["user_id"]
    messages = state["messages"]
    
    # 提取用户提到的信息（简化版，实际应该用 LLM 提取）
    last_user_message = None
    for msg in messages:
        if msg.type == "human":
            last_user_message = msg.content
    
    if last_user_message:
        # 假设用户说"我叫 Alice"，提取名字
        if "我叫" in last_user_message:
            name = last_user_message.split("我叫")[1].split()[0]
            store.put(("user", user_id, "profile"), "name", {"name": name})
    
    return {}
```

### 第三步：构建图

```python
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import MemorySaver
from langgraph.store.memory import InMemoryStore
from typing import TypedDict
from langchain_core.messages import BaseMessage

class AgentState(TypedDict):
    user_id: str
    messages: list[BaseMessage]
    faq_answer: str

store = InMemoryStore()
checkpointer = MemorySaver()

graph = StateGraph(AgentState)

graph.add_node("greet", greet_node)
graph.add_node("faq_match", faq_match_node)
graph.add_node("llm", llm_node)
graph.add_node("tools", tool_node)
graph.add_node("save_memory", save_memory_node)

# 流程
graph.add_edge(START, "greet")
graph.add_edge("greet", "faq_match")
graph.add_edge("faq_match", "llm")

# 条件路由：LLM 是否需要工具？
def should_use_tools(state):
    last_message = state["messages"][-1]
    if hasattr(last_message, "tool_calls") and last_message.tool_calls:
        return "tools"
    return "end"

graph.add_conditional_edges("llm", should_use_tools, {
    "tools": "tools",
    "end": "save_memory"
})

graph.add_edge("tools", "llm")  # 工具执行后回到 LLM
graph.add_edge("save_memory", END)

app = graph.compile(checkpointer=checkpointer)
```

### 第四步：测试

```python
def chat(user_id: str, user_message: str, thread_id: str):
    result = app.invoke(
        {
            "user_id": user_id,
            "messages": [{"role": "user", "content": user_message}]
        },
        config={"configurable": {"thread_id": thread_id}}
    )
    return result["messages"][-1].content

# 测试 1：问候（有用户偏好）
store.put(("user", "alice", "preferences"), "language", {"value": "中文"})
store.put(("user", "alice", "preferences"), "theme", {"value": "深色模式"})

response = chat("alice", "你好", "thread_1")
print(f"AI: {response}")
# 输出：欢迎回来！根据您的偏好，使用 中文 为您服务。

# 测试 2：FAQ 问题
response = chat("alice", "如何退款？", "thread_1")
print(f"AI: {response}")
# 输出：申请退款请联系客服，提供订单号。一般 3-5 个工作日处理。

# 测试 3：订单查询（需要工具）
response = chat("alice", "帮我查一下订单 1001", "thread_1")

# 输出：订单 1001（iPhone 15）状态：已发货

# 测试 4：跨对话记忆
response = chat("alice", "我叫 Bob", "thread_2")  # 新 thread
print(f"AI: {response}")

# 再次调用，应该记得名字叫 Bob

print(f"AI: {response}")
```

## 扩展方向

这个客服 Agent 还可以扩展：

### 1. 更智能的用户记忆提取

用 LLM 自动提取用户提到的信息：

```python
@tool
def extract_user_info(messages: list) -> dict:
    """从对话中提取用户信息"""
    response = llm.invoke([
        SystemMessage(content="从对话中提取用户信息（姓名、地址、偏好等）"),
        *messages
    ])
    # 解析 LLM 的输出
    return {...}
```

### 2. 多语言支持

根据用户的语言偏好，用对应语言回答：

```python
def llm_node(state):
    user_id = state["user_id"]
    prefs = store.search(("user", user_id, "preferences"))
    language = next((p.value["value"] for p in prefs if p.key == "language"), "中文")
    
    system_prompt = f"请使用 {language} 回答用户的问题。"
    ...
```

### 3. 持久化 Store

生产环境使用 PostgresStore：

```python
from langgraph.store.postgres import PostgresStore

store = PostgresStore.from_conn_string("postgresql://...")
```

### 4. 并发支持

使用异步 API 支持多用户并发：

```python
async def chat_async(user_id: str, user_message: str, thread_id: str):
    result = await app.ainvoke(...)
    return result["messages"][-1].content
```

## 小结

通过这个项目，我们综合运用了 LangGraph 的核心特性：

1. **Checkpointer**：保存对话历史，支持多轮对话
2. **Store**：保存用户偏好、FAQ 等长期记忆
3. **StateGraph**：构建复杂的工作流（问候 → FAQ 匹配 → LLM → 工具 → 保存记忆）
4. **工具调用**：集成搜索、订单查询等能力

这个客服 Agent 具备了：
- ✅ 多轮对话能力
- ✅ 长期记忆能力
- ✅ 个性化服务
- ✅ 工具调用能力

掌握了这些特性，你就能构建各种实用的 Agent 应用了！
