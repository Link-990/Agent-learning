# 6.5 三种操作：approve、edit、reject

当 Agent 暂停等待人工输入时，人类通常有 3 种选择：

1. **approve（批准）**：直接通过，继续执行
2. **edit（编辑）**：修改内容，然后继续
3. **reject（拒绝）**：拒绝，终止或回退

这一节我们学习如何在代码中处理这 3 种操作。

---

## 处理 approve

最简单的情况——人类批准，直接继续。

```python
@entrypoint(checkpointer=MemorySaver())
def workflow(task: str):
    plan = f"执行：{task}"
    
    # 等待审批
    decision = interrupt({"plan": plan, "options": ["approve", "reject"]})
    
    if decision == "approve":
        return f"✅ 已批准：{plan}"
    else:
        return f"❌ 已拒绝：{plan}"

# 人类批准
result = workflow.invoke(Command(resume="approve"), config={"thread_id": "1"})
```

---

## 处理 reject

拒绝的情况——终止或回退。

```python
@entrypoint(checkpointer=MemorySaver())
def workflow(task: str):
    plan = f"执行：{task}"
    
    decision = interrupt({"plan": plan, "options": ["approve", "reject"]})
    
    if decision == "reject":
        # 可以抛出异常，或者返回错误信息
        raise ValueError(f"计划被拒绝：{plan}")
    
    return f"✅ 已批准：{plan}"

# 人类拒绝
try:
    workflow.invoke(Command(resume="reject"), config={"thread_id": "1"})
except ValueError as e:
    print(e)  # "计划被拒绝：执行：..."
```

---

## 处理 edit

编辑的情况——修改内容，然后继续。

```python
@entrypoint(checkpointer=MemorySaver())
def workflow(task: str):
    plan = f"执行：{task}"
    
    decision = interrupt({"plan": plan, "options": ["approve", "edit", "reject"]})
    
    if decision["action"] == "approve":
        return f"✅ {plan}"
    elif decision["action"] == "edit":
        # 人类提供了修改后的版本
        modified_plan = decision["modified_plan"]
        return f"✅ 已修改并批准：{modified_plan}"
    else:
        return f"❌ 已拒绝"

# 人类编辑
result = workflow.invoke(
    Command(resume={
        "action": "edit",
        "modified_plan": "执行：修改后的任务"
    }),
    config={"thread_id": "1"}
)
```

## 完整示例：带 3 种操作的审批流

```python
from langgraph.func import entrypoint, interrupt
from langgraph.checkpoint.memory import MemorySaver
from langgraph.types import Command

@entrypoint(checkpointer=MemorySaver())
def approval_workflow(task: str, budget: float):
    """
    审批工作流，支持 3 种操作：
    1. approve：批准
    2. edit：修改预算后批准
    3. reject：拒绝
    """
    plan = {
        "task": task,
        "budget": budget
    }
    
    # 暂停，等待审批
    decision = interrupt({
        "plan": plan,
        "options": ["approve", "edit", "reject"],
        "message": "请审批此计划"
    })
    
    action = decision.get("action")
    
    if action == "approve":
        return {
            "status": "approved",
            "plan": plan,
            "message": "计划已批准"
        }
    
    elif action == "edit":
        # 修改预算
        new_budget = decision.get("new_budget", budget)
        modified_plan = {**plan, "budget": new_budget}
        return {
            "status": "approved",
            "plan": modified_plan,
            "message": f"预算已修改为 {new_budget} 元，计划已批准"
        }
    
    elif action == "reject":
        reason = decision.get("reason", "无")
        return {
            "status": "rejected",
            "plan": plan,
            "message": f"计划被拒绝，原因：{reason}"
        }
    
    else:
        return {
            "status": "error",
            "message": f"未知操作：{action}"
        }

# 测试 1：批准
thread_id_1 = "test_1"
approval_workflow("建设大楼", 100000, config={"thread_id": thread_id_1})
result = approval_workflow.invoke(
    Command(resume={"action": "approve"}),
    config={"thread_id": thread_id_1}
)
print(f"测试 1：{result}")
# 输出：{'status': 'approved', 'plan': {...}, 'message': '计划已批准'}

# 测试 2：编辑（修改预算）
thread_id_2 = "test_2"
approval_workflow("购买设备", 50000, config={"thread_id": thread_id_2})
result = approval_workflow.invoke(
    Command(resume={"action": "edit", "new_budget": 30000}),
    config={"thread_id": thread_id_2}
)
print(f"测试 2：{result}")
# 输出：{'status': 'approved', 'plan': {..., 'budget': 30000}, 'message': '预算已修改为 30000 元，计划已批准'}

# 测试 3：拒绝
thread_id_3 = "test_3"
approval_workflow("团建活动", 20000, config={"thread_id": thread_id_3})
result = approval_workflow.invoke(
    Command(resume={"action": "reject", "reason": "预算超标"}),
    config={"thread_id": thread_id_3}
)
print(f"测试 3：{result}")
# 输出：{'status': 'rejected', 'plan': {...}, 'message': '计划被拒绝，原因：预算超标'}
```

## 编辑模式的进阶用法

有时候，编辑可能需要多轮交互：

```python
@entrypoint(checkpointer=MemorySaver())
def iterative_edit_workflow(content: str):
    """
    迭代编辑工作流：
    人类可以多次编辑，直到满意
    """
    draft = content
    
    while True:
        # 展示草稿，等待审批
        decision = interrupt({
            "draft": draft,
            "options": ["approve", "edit", "reject"]
        })
        
        action = decision.get("action")
        
        if action == "approve":
            return {"status": "approved", "final_draft": draft}
        
        elif action == "edit":
            # 人类提供修改意见
            feedback = decision.get("feedback")
            # 这里可以用 LLM 根据反馈修改草稿
            # 简化版：直接替换
            draft = f"{draft}\n\n修订版：{feedback}"
            # 继续循环，再次展示
        
        elif action == "reject":
            return {"status": "rejected", "reason": decision.get("reason")}

# 测试
thread_id = "iterative_1"

# 第一次：等待审批
iterative_edit_workflow("原始内容", config={"thread_id": thread_id})

# 第一次编辑
iterative_edit_workflow.invoke(
    Command(resume={"action": "edit", "feedback": "第一段需要更详细"}),
    config={"thread_id": thread_id}
)

# 第二次编辑
iterative_edit_workflow.invoke(
    Command(resume={"action": "edit", "feedback": "语气太正式，改轻松一些"}),
    config={"thread_id": thread_id}
)

# 最终批准
result = iterative_edit_workflow.invoke(
    Command(resume={"action": "approve"}),
    config={"thread_id": thread_id}
)
print(result)
```

## StateGraph API 中的 3 种操作

在 StateGraph 中，可以用节点来处理不同的操作：

```python
from langgraph.graph import StateGraph, START, END
from typing import TypedDict

class State(TypedDict):
    content: str
    action: str  # "approve", "edit", "reject"
    feedback: str

def approve_node(state: State):
    return {"status": "approved", "content": state["content"]}

def edit_node(state: State):
    # 根据 feedback 修改内容
    modified = f"{state['content']}\n\n修订：{state['feedback']}"
    return {"status": "edited", "content": modified}

def reject_node(state: State):
    return {"status": "rejected"}

def route_decision(state: State):
    action = state.get("action")
    if action == "approve":
        return "approve"
    elif action == "edit":
        return "edit"
    else:
        return "reject"

graph = StateGraph(State)
graph.add_node("approve", approve_node)
graph.add_node("edit", edit_node)
graph.add_node("reject", reject_node)

graph.add_edge(START, "decision")
graph.add_conditional_edges("decision", route_decision, {
    "approve": "approve",
    "edit": "edit",
    "reject": "reject"
})
graph.add_edge("approve", END)
graph.add_edge("edit", END)
graph.add_edge("reject", END)

# 在 decision 节点之前暂停
app = graph.compile(
    checkpointer=MemorySaver(),
    interrupt_before=["decision"]
)
```

## 小结

3 种操作的处理：

1. **approve（批准）**：直接继续
   ```python
   Command(resume={"action": "approve"})
   ```

2. **edit（编辑）**：修改后继续
   ```python
   Command(resume={"action": "edit", "new_value": ...})
   ```

3. **reject（拒绝）**：终止或回退
   ```python
   Command(resume={"action": "reject", "reason": ...})
   ```

**最佳实践**：
- 用字典传递决策（包含 action 和其他字段）
- 编辑操作可以支持多轮迭代
- 拒绝操作可以提供原因
