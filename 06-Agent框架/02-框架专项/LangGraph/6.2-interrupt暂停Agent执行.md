# 6.2 interrupt()：暂停 Agent 执行

`interrupt()` 是 LangGraph 提供的**函数式暂停机制**——在代码的任意位置调用它，Agent 就会暂停，等待人类输入。

---

## 基本用法

### 1. 导入

```python
from langgraph.func import interrupt
```

### 2. 在代码中调用

```python
@entrypoint(checkpointer=MemorySaver())
def workflow(topic: str):
    plan = f"计划：{topic}"
    
    # 暂停，等待人类决策
    approved = interrupt({"plan": plan, "question": "是否批准此计划？"})
    
    if approved:
        return f"执行：{plan}"
    else:
        return "计划被拒绝"
```

### 3. 调用工作流

```python
# 第一次调用：会在 interrupt 处暂停
try:
    result = workflow("建设大楼", config={"thread_id": "1"})
except:
    print("工作流已暂停，等待审批")
```

### 4. 恢复执行

```python
from langgraph.types import Command

# 人类审批后，恢复执行
result = workflow.invoke(Command(resume=True), config={"thread_id": "1"})
print(result)  # "执行：计划：建设大楼"
```

---

## interrupt 的返回值

`interrupt()` 会返回**人类传入的值**。

```python
# 暂停，等待人类输入
human_input = interrupt({"question": "请输入你的名字："})

# human_input 就是人类传入的值
return f"你好，{human_input}！"
```

人类可以通过 `Command(resume=value)` 传入任意值：

```python
# 传入字符串
workflow.invoke(Command(resume="Alice"), config={"thread_id": "1"})

# 传入数字
workflow.invoke(Command(resume=42), config={"thread_id": "1"})

# 传入字典
workflow.invoke(Command(resume={"name": "Alice", "age": 25}), config={"thread_id": "1"})
```

---

## interrupt 的参数

`interrupt()` 接收一个参数：**要传递给人类的信息**。

```python
interrupt(payload)
```

这个 `payload` 可以是任何 Python 对象：

```python
# 字符串
interrupt("请输入你的名字")

# 字典
interrupt({
    "question": "是否批准？",
    "plan": plan,
    "cost": 10000
})

# 自定义对象
interrupt(MyDataClass(field1="value1", field2="value2"))
```

通常用字典，因为可以包含多个字段，方便人类理解上下文。

## 完整的审批流程示例

```python
from dotenv import load_dotenv
from langgraph.func import entrypoint, interrupt
from langgraph.checkpoint.memory import MemorySaver
from langgraph.types import Command

load_dotenv()

checkpointer = MemorySaver()

@entrypoint(checkpointer=checkpointer)
def approval_workflow(task: str, cost: float):
    """
    审批工作流：
    1. 生成执行计划
    2. 暂停，等待人类审批
    3. 根据审批结果决定
    """
    # 1. 生成计划
    plan = f"执行任务：{task}，预算：{cost} 元"
    
    # 2. 暂停，等待审批
    approval = interrupt({
        "plan": plan,
        "cost": cost,
        "message": "请审批此计划（回复 True 批准，False 拒绝）"
    })
    
    # 3. 根据审批结果
    if approval:
        return f"✅ 已批准：{plan}"
    else:
        return f"❌ 已拒绝：{plan}"

# 测试
thread_id = "approval_1"

# 第一次调用：会暂停
print("=== 第一次调用 ===")
try:
    workflow("建设新办公室", 50000)
except Exception as e:
    print(f"工作流已暂停：{e}")

# 人工审批（假设批准）
print("\n=== 人工审批（批准）===")
result = approval_workflow.invoke(Command(resume=True), config={"thread_id": thread_id})
print(result)
# 输出：✅ 已批准：执行任务：建设新办公室，预算：50000 元

# 重新测试：这次拒绝
thread_id_2 = "approval_2"
print("\n=== 第二次调用 ===")
try:
    approval_workflow("购买奢侈品", 100000, config={"thread_id": thread_id_2})
except Exception as e:
    print(f"工作流已暂停：{e}")

# 人工拒绝
print("\n=== 人工审批（拒绝）===")
result = approval_workflow.invoke(Command(resume=False), config={"thread_id": thread_id_2})
print(result)
# 输出：❌ 已拒绝：执行任务：购买奢侈品，预算：100000 元
```

## 多次 interrupt

一个工作流可以有多个 `interrupt`，每次暂停都可以等待不同的决策。

### 示例：多步骤审批

```python
@entrypoint(checkpointer=MemorySaver())
def multi_step_approval(project: str):
    # 步骤 1：技术审批
    tech_approved = interrupt({
        "step": "技术审批",
        "project": project,
        "message": "技术团队是否批准？"
    })
    
    if not tech_approved:
        return "❌ 技术团队未批准"
    
    # 步骤 2：财务审批
    budget = interrupt({
        "step": "财务审批",
        "project": project,
        "message": "请指定预算金额："
    })
    
    finance_approved = interrupt({
        "step": "财务审批",
        "project": project,
        "budget": budget,
        "message": "财务部门是否批准？"
    })
    
    if not finance_approved:
        return f"❌ 财务部门未批准预算 {budget} 元"
    
    return f"✅ 项目 {project} 已批准，预算 {budget} 元"

# 测试
thread_id = "multi_1"

# 第一次：技术审批
print("=== 技术审批 ===")
multi_step_approval("AI 系统升级", config={"thread_id": thread_id})

# 恢复：技术批准
print("\n=== 恢复：技术批准 ===")
multi_step_approval.invoke(Command(resume=True), config={"thread_id": thread_id})

# 第二次暂停：指定预算
print("\n=== 指定预算 ===")
# 继续执行...

# 恢复：指定预算 100000
multi_step_approval.invoke(Command(resume=100000), config={"thread_id": thread_id})

# 第三次暂停：财务审批
# 恢复：财务批准
result = multi_step_approval.invoke(Command(resume=True), config={"thread_id": thread_id})
print(result)
# 输出：✅ 项目 AI 系统升级 已批准，预算 100000 元
```

## interrupt 的暂停点

当调用 `interrupt()` 时，LangGraph 会：

1. **保存当前状态**：通过 Checkpointer
2. **暂停执行**：函数停止运行
3. **等待人类输入**：直到调用 `invoke(Command(resume=...))`

恢复时：
1. **加载状态**：从 Checkpointer 读取
2. **从暂停点继续**：`interrupt()` 返回人类传入的值
3. **继续执行**：函数的剩余部分

---

## 注意事项

### 1. 必须配置 Checkpointer

```python
@entrypoint  # ❌ 没有 checkpointer
def workflow():
    interrupt(...)  # 运行时报错
```

```python
@entrypoint(checkpointer=MemorySaver())  # ✅ 配置了
def workflow():
    interrupt(...)  # 正常工作
```

### 2. interrupt 会抛出异常

调用包含 `interrupt` 的工作流时，会抛出 `GraphInterrupt` 异常：

```python
try:
    workflow("test")
except GraphInterrupt as e:
    print("工作流已暂停，等待审批")
```

### 3. resume 的值可以是任意类型

```python
# 布尔值
Command(resume=True)

# 字符串
Command(resume="Alice")

# 数字
Command(resume=42)

# 字典
Command(resume={"approved": True, "comment": "看起来不错"})

# 列表
Command(resume=[1, 2, 3])
```

## 实战：一个带输入的工作流

```python
@entrypoint(checkpointer=MemorySaver())
def interactive_workflow(topic: str):
    # 1. 生成大纲
    outline = f"{topic} 的大纲"
    
    # 2. 让用户选择风格
    style = interrupt({
        "message": "请选择文章风格：",
        "options": ["正式", "轻松", "幽默"]
    })
    
    # 3. 根据风格生成文章
    article = f"基于《{outline}》，用{style}风格写的文章..."
    
    # 4. 让用户审阅
    feedback = interrupt({
        "message": "请审阅文章：",
        "article": article,
        "options": ["满意", "需要修改", "重写"]
    })
    
    # 5. 根据反馈决定
    if feedback == "满意":
        return article
    elif feedback == "需要修改":
        # 等待修改意见
        revision = interrupt({"message": "请输入修改意见："})
        return f"{article}\n\n修订版：{revision}"
    else:
        # 重写
        new_article = f"重新写的文章：{topic}"
        return new_article

# 测试
thread_id = "interactive_1"

# 第一次：等待选择风格
interactive_workflow("人工智能", config={"thread_id": thread_id})

# 恢复：选择"幽默"风格
interactive_workflow.invoke(Command(resume="幽默"), config={"thread_id": thread_id})

# 第二次：等待审阅
# 恢复：选择"满意"
result = interactive_workflow.invoke(Command(resume="满意"), config={"thread_id": thread_id})
print(result)
```

---

## 小结

`interrupt()` 的用法：

**暂停**：
```python
value = interrupt(payload)  # payload 传递给人类的信息
```

**恢复**：
```python
workflow.invoke(Command(resume=human_value), config={"thread_id": "..."})
```

**关键点**：
- `interrupt()` 会返回人类传入的值
- 必须配置 Checkpointer
- 一个工作流可以有多个 interrupt
- `resume` 的值可以是任意类型

掌握了 `interrupt()`，你就能在 Agent 的任意位置添加人工介入点，让它真正"可控"。
