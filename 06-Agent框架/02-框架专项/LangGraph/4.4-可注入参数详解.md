# 4.4 可注入参数详解

在 Functional API 中，有一些"魔法参数"——你在函数签名里声明它们，LangGraph 会自动注入对应的值。

这些参数让你能够访问运行时的上下文、Store 系统、流式输出等高级功能。

这一节我们详细学习这些可注入参数。

---

## 可注入参数列表

| 参数名 | 类型 | 作用 |
|--------|------|------|
| `previous` | 任意 | 上一个 task 的返回值 |
| `store` | Store | Store 系统（长期记忆） |
| `writer` | StreamWriter | 流式输出 |
| `config` | RunnableConfig | 运行时配置 |

---

## 1. previous：上一步的结果

`previous` 参数让你获取**上一个 task 的返回值**。

### 基本用法

```python
from langgraph.func import entrypoint, task

@task
def step1() -> str:
    return "第一步的结果"

@task
def step2(previous) -> str:
    # previous 是 step1 的返回值
    return f"收到：{previous}"

@entrypoint
def workflow() -> str:
    step1()
    result = step2()
    return result

print(workflow())
# 输出：收到：第一步的结果
```

### 链式调用

```python
@task
def add_10(value: int) -> int:
    return value + 10

@task
def multiply_2(previous: int) -> int:
    return previous * 2

@task
def subtract_5(previous: int) -> int:
    return previous - 5

@entrypoint
def workflow(value: int) -> int:
    add_10(value)
    multiply_2()
    result = subtract_5()
    return result

print(workflow(5))
# 5 + 10 = 15 → 15 * 2 = 30 → 30 - 5 = 25
```

### previous 的类型标注

```python
@task
def step1() -> dict:
    return {"key": "value"}

@task
def step2(previous: dict) -> str:
    return previous["key"]
```

IDE 会根据类型标注做检查和自动补全。

## 2. store：长期记忆

`store` 参数让你访问 Store 系统——跨线程、跨会话的长期记忆。

### 配置 store

```python
from langgraph.func import entrypoint, task
from langgraph.store.memory import InMemoryStore

store = InMemoryStore()

@entrypoint(store=store)
def workflow(user_id: str) -> str:
    ...
```

### 在 task 里使用 store

```python
@task
def save_preference(user_id: str, preference: str, store):
    # 保存用户偏好
    store.put(
        namespace=("user", user_id),
        key="preference",
        value={"preference": preference}
    )
    return "已保存"

@task
def load_preference(user_id: str, store) -> str:
    # 读取用户偏好
    items = store.search(namespace=("user", user_id))
    if items:
        return items[0].value.get("preference", "无")
    return "无"

@entrypoint(store=store)
def workflow(user_id: str, action: str, preference: str = None) -> str:
    if action == "save":
        return save_preference(user_id, preference)
    else:
        return load_preference(user_id)

# 保存
workflow("user_123", "save", "dark_mode")

# 读取
result = workflow("user_123", "load")
print(result)  # "dark_mode"
```

### store 的 API

- `store.put(namespace, key, value)`：保存数据
- `store.get(namespace, key)`：获取单条数据
- `store.search(namespace, filter, limit, offset)`：搜索数据
- `store.delete(namespace, key)`：删除数据

详细用法会在"持久化系统"章节讲解。

## 3. writer：流式输出

`writer` 参数让你在 task 执行过程中，实时输出中间结果。

### 基本用法

```python
from langgraph.func import entrypoint, task

@task
def long_task(topic: str, writer):
    writer.write(f"开始处理：{topic}")
    
    # 步骤 1
    writer.write("生成大纲...")
    outline = f"{topic} 的大纲"
    
    # 步骤 2
    writer.write("撰写内容...")
    content = f"基于 {outline} 的内容"
    
    writer.write("完成！")
    return content

@entrypoint
def workflow(topic: str) -> str:
    result = long_task(topic)
    return result

# 流式执行
for event in workflow.stream("AI"):
    print(event)
```

输出：

```
{'long_task': {'messages': ['开始处理：AI']}}
{'long_task': {'messages': ['生成大纲...']}}
{'long_task': {'messages': ['撰写内容...']}}
{'long_task': {'messages': ['完成！']}}
{'long_task': '基于 AI 的大纲 的内容'}
```

### writer 的 API

- `writer.write(content)`：输出一条消息

流式输出的详细用法会在"流式输出"章节讲解。

## 4. config：运行时配置

`config` 参数让你访问运行时配置——比如 `thread_id`、自定义配置等。

### 基本用法

```python
from langgraph.func import entrypoint, task
from langgraph.types import RunnableConfig

@task
def my_task(input: str, config: RunnableConfig) -> str:
    thread_id = config.get("configurable", {}).get("thread_id")
    return f"处理 {input}，线程 {thread_id}"

@entrypoint
def workflow(input: str) -> str:
    result = my_task(input)
    return result

result = workflow("hello", config={"configurable": {"thread_id": "thread_1"}})
print(result)
# 输出：处理 hello，线程 thread_1
```

### config 的结构

```python
{
    "configurable": {
        "thread_id": "thread_1",
        "checkpoint_id": "...",
        # 自定义配置
        "user_id": "user_123",
        "session_id": "session_456"
    },
    "metadata": { ... },
    "tags": [ ... ]
}
```

### 传递自定义配置

```python
@task
def personalized_task(input: str, config: RunnableConfig) -> str:
    user_id = config.get("configurable", {}).get("user_id")
    return f"为用户 {user_id} 处理 {input}"

@entrypoint
def workflow(input: str) -> str:
    result = personalized_task(input)
    return result

result = workflow(
    "hello",
    config={"configurable": {"user_id": "Alice"}}
)
print(result)
# 输出：为用户 Alice 处理 hello
```

## 混合使用多个可注入参数

你可以在同一个 task 里使用多个可注入参数：

```python
@task
def complex_task(
    input: str,
    previous: dict,
    store,
    writer,
    config: RunnableConfig
) -> str:
    # 1. 获取运行时配置
    user_id = config.get("configurable", {}).get("user_id")
    
    # 2. 从 store 读取用户记忆
    memories = store.search(("user", user_id))
    
    # 3. 使用上一步的结果
    prev_result = previous.get("result")
    
    # 4. 流式输出进度
    writer.write(f"为用户 {user_id} 处理 {input}")
    
    # 5. 业务逻辑
    result = f"{prev_result} + {input} + {len(memories)} 条记忆"
    
    return result
```

## 可注入参数的识别规则

LangGraph 如何知道一个参数是"可注入参数"还是"普通参数"？

### 规则

1. **参数名匹配**：如果参数名是 `previous`、`store`、`writer`、`config`，就是可注入参数
2. **类型标注**（可选）：可以加类型标注，但不是必须的

### 示例

```python
# 这些都是可注入参数
def task1(previous):          # ✅ 参数名是 previous
def task2(store):             # ✅ 参数名是 store
def task3(writer):            # ✅ 参数名是 writer
def task4(config):            # ✅ 参数名是 config

# 加上类型标注更好
def task5(previous: dict):    # ✅ 更清晰
def task6(store: Store):      # ✅ 更清晰
def task7(config: RunnableConfig):  # ✅ 更清晰
```

### 普通参数 vs 可注入参数

```python
@task
def my_task(
    input: str,      # 普通参数：调用时传入
    count: int,      # 普通参数：调用时传入
    previous: dict,  # 可注入参数：LangGraph 自动注入
    store            # 可注入参数：LangGraph 自动注入
) -> str:
    ...

# 调用时只需要传普通参数
my_task("hello", 10)
```

## 可注入参数的限制

### 1. 只能在 @entrypoint 和 @task 里使用

```python
def normal_function(previous):  # ❌ 不会注入，previous 是 undefined
    return previous

@task
def my_task(previous):  # ✅ 会注入
    return previous
```

### 2. store 需要在 @entrypoint 里配置

```python
@entrypoint  # ❌ 没有配置 store
def workflow():
    my_task()  # task 里用了 store，会报错

@entrypoint(store=InMemoryStore())  # ✅ 配置了 store
def workflow():
    my_task()  # 可以用 store
```

## 实战示例：一个带记忆的对话 Agent

```python
from langgraph.func import entrypoint, task
from langgraph.store.memory import InMemoryStore
from langgraph.checkpoint.memory import MemorySaver

store = InMemoryStore()
checkpointer = MemorySaver()

@task
def load_history(user_id: str, store) -> list:
    """加载历史对话"""
    items = store.search(("user", user_id, "history"))
    if items:
        return items[0].value.get("messages", [])
    return []

@task
def chat(user_input: str, previous: list, writer) -> str:
    """调用 LLM 聊天"""
    history = previous  # 上一步加载的历史
    
    writer.write("思考中...")
    
    # 这里调用 LLM（简化）
    response = f"回复：{user_input}"
    
    writer.write("回复完成")
    return response

@task
def save_history(user_id: str, user_input: str, ai_response: str, previous: list, store):
    """保存对话历史"""
    history = previous  # 之前的历史
    history.append({"role": "user", "content": user_input})
    history.append({"role": "ai", "content": ai_response})
    
    store.put(
        ("user", user_id, "history"),
        "messages",
        {"messages": history}
    )

@entrypoint(store=store, checkpointer=checkpointer)
def chat_workflow(user_id: str, user_input: str, config) -> str:
    # 1. 加载历史
    load_history(user_id)
    
    # 2. 聊天
    response = chat(user_input)
    
    # 3. 保存历史
    save_history(user_id, user_input, response)
    
    return response

# 第一轮对话
result = chat_workflow("user_123", "你好", config={"thread_id": "1"})
print(result)

# 第二轮对话（会加载之前的历史）
result = chat_workflow("user_123", "天气怎么样", config={"thread_id": "2"})
print(result)
```

## 小结

Functional API 的可注入参数：

1. **previous**：上一个 task 的返回值
   - 用于链式调用
   
2. **store**：Store 系统
   - 跨线程、跨会话的长期记忆
   - 需要在 @entrypoint 配置
   
3. **writer**：流式输出
   - 实时输出中间结果
   - 用 `writer.write()` 输出
   
4. **config**：运行时配置
   - 获取 thread_id、自定义配置等

识别规则：参数名匹配 + 可选的类型标注

这些可注入参数让 Functional API 能够访问 LangGraph 的高级特性，而不需要显式传递。
