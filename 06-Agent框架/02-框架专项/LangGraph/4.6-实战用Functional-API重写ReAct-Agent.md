# 4.6 实战：用 Functional API 重写 ReAct Agent

在第 3.7 节，我们用 StateGraph API 实现了一个 ReAct Agent。现在，我们用 Functional API 重写它，对比两种方式的区别。

---

## 回顾：StateGraph 版本

```python
from langgraph.graph import StateGraph, MessagesState, START, END
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool

llm = ChatOpenAI(model="gpt-4o-mini")

@tool
def search(query: str) -> str:
    """搜索工具"""
    return f"搜索结果：{query}"

tools = [search]
llm_with_tools = llm.bind_tools(tools)

def call_llm(state: MessagesState):
    response = llm_with_tools.invoke(state["messages"])
    return {"messages": [response]}

def call_tools(state: MessagesState):
    last_message = state["messages"][-1]
    tool_calls = last_message.tool_calls
    tool_map = {tool.name: tool for tool in tools}
    
    results = []
    for tool_call in tool_calls:
        tool_func = tool_map[tool_call["name"]]
        result = tool_func.invoke(tool_call["args"])
        results.append(ToolMessage(content=result, tool_call_id=tool_call["id"]))
    
    return {"messages": results}

def should_continue(state: MessagesState):
    last_message = state["messages"][-1]
    if hasattr(last_message, "tool_calls") and last_message.tool_calls:
        return "tools"
    return "end"

graph = StateGraph(MessagesState)
graph.add_node("llm", call_llm)
graph.add_node("tools", call_tools)
graph.add_edge(START, "llm")
graph.add_conditional_edges("llm", should_continue, {"tools": "tools", "end": END})
graph.add_edge("tools", "llm")

app = graph.compile()
```

## Functional API 版本

### 方式 1：最简洁版

```python
from langgraph.func import entrypoint, task
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.messages import ToolMessage

llm = ChatOpenAI(model="gpt-4o-mini")

@tool
def search(query: str) -> str:
    """搜索工具"""
    return f"搜索结果：{query}"

@tool
def calculator(expression: str) -> str:
    """计算器工具"""
    try:
        return str(eval(expression))
    except:
        return "计算错误"

tools = [search, calculator]
llm_with_tools = llm.bind_tools(tools)
tool_map = {tool.name: tool for tool in tools}

@entrypoint
def react_agent(messages: list) -> str:
    """ReAct Agent"""
    while True:
        # 调用 LLM
        response = llm_with_tools.invoke(messages)
        messages.append(response)
        
        # 判断是否需要工具
        if not hasattr(response, "tool_calls") or not response.tool_calls:
            # 不需要工具，返回最终答案
            return response.content
        
        # 执行工具调用
        for tool_call in response.tool_calls:
            tool_func = tool_map[tool_call["name"]]
            result = tool_func.invoke(tool_call["args"])
            
            # 把工具结果加入消息列表
            messages.append(
                ToolMessage(content=result, tool_call_id=tool_call["id"])
            )

# 测试
result = react_agent([{"role": "user", "content": "搜索：Python 教程"}])
print(result)
```

**代码量：约 30 行**（StateGraph 版本是 45 行）

### 方式 2：用 @task 拆分

如果想让每一步都是独立的 task（方便追踪、暂停），可以这样写：

```python
from langgraph.func import entrypoint, task
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.messages import ToolMessage

llm = ChatOpenAI(model="gpt-4o-mini")

@tool
def search(query: str) -> str:
    """搜索工具"""
    return f"搜索结果：{query}"

tools = [search]
llm_with_tools = llm.bind_tools(tools)
tool_map = {tool.name: tool for tool in tools}

@task
def call_llm(messages: list, writer):
    """调用 LLM"""
    writer.write("正在思考...")
    response = llm_with_tools.invoke(messages)
    return response

@task
def execute_tools(tool_calls, writer):
    """执行工具调用"""
    writer.write(f"正在调用 {len(tool_calls)} 个工具...")
    results = []
    for tool_call in tool_calls:
        tool_func = tool_map[tool_call["name"]]
        result = tool_func.invoke(tool_call["args"])
        results.append(
            ToolMessage(content=result, tool_call_id=tool_call["id"])
        )
    return results

@entrypoint
def react_agent(messages: list) -> str:
    """ReAct Agent"""
    while True:
        # 1. 调用 LLM
        response = call_llm(messages)
        messages.append(response)
        
        # 2. 判断是否需要工具
        if not hasattr(response, "tool_calls") or not response.tool_calls:
            return response.content
        
        # 3. 执行工具
        tool_results = execute_tools(response.tool_calls)
        messages.extend(tool_results)

# 流式测试
for event in react_agent.stream([{"role": "user", "content": "搜索：AI"}]):
    print(event)
```

## 完整示例：带多个工具的 Agent

```python
from dotenv import load_dotenv
from langgraph.func import entrypoint
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.messages import ToolMessage
from datetime import datetime

load_dotenv()

# ========== 定义工具 ==========
@tool
def search(query: str) -> str:
    """搜索互联网信息"""
    fake_results = {
        "天气": "今天晴天，气温 25°C",
        "python": "Python 是一种高级编程语言",
        "ai": "AI 是人工智能的缩写"
    }
    for key, value in fake_results.items():
        if key in query.lower():
            return value
    return f"未找到关于 '{query}' 的信息"

@tool
def calculator(expression: str) -> str:
    """计算数学表达式"""
    try:
        result = eval(expression)
        return str(result)
    except Exception as e:
        return f"计算错误：{str(e)}"

@tool
def get_time() -> str:
    """获取当前时间"""
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")

# ========== 绑定工具 ==========
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
tools = [search, calculator, get_time]
llm_with_tools = llm.bind_tools(tools)
tool_map = {tool.name: tool for tool in tools}

# ========== Agent 实现 ==========
@entrypoint
def react_agent(user_input: str) -> str:
    """
    ReAct Agent：能够自主调用工具的智能助手
    
    工作流程：
    1. 接收用户输入
    2. 调用 LLM 分析是否需要工具
    3. 如果需要，执行工具并返回结果给 LLM
    4. 重复 2-3，直到 LLM 给出最终答案
    """
    messages = [{"role": "user", "content": user_input}]
    
    max_iterations = 10  # 防止无限循环
    iteration = 0
    
    while iteration < max_iterations:
        iteration += 1
        
        # 调用 LLM
        response = llm_with_tools.invoke(messages)
        messages.append(response)
        
        # 判断是否需要工具
        if not hasattr(response, "tool_calls") or not response.tool_calls:
            # 不需要工具，返回最终答案
            return response.content
        
        # 执行工具调用
        for tool_call in response.tool_calls:
            tool_name = tool_call["name"]
            tool_args = tool_call["args"]
            
            # 调用工具
            tool_func = tool_map[tool_name]
            result = tool_func.invoke(tool_args)
            
            # 把工具结果加入消息列表
            messages.append(
                ToolMessage(
                    content=result,
                    tool_call_id=tool_call["id"]
                )
            )
    
    return "抱歉，达到了最大迭代次数，无法完成任务。"

# ========== 测试 ==========
if __name__ == "__main__":
    test_queries = [
        "今天天气怎么样？",
        "123 乘以 456 等于多少？",
        "现在几点了？",
        "你好，介绍一下 AI",
        "先告诉我现在几点，然后计算 100 加 200"
    ]
    
    for query in test_queries:
        print(f"\n{'='*60}")
        print(f"用户: {query}")
        print(f"{'='*60}")
        result = react_agent(query)
        print(f"AI: {result}")
```

## 对比：StateGraph vs Functional

### 代码量

| 版本 | 代码行数 |
|------|---------|
| StateGraph | ~65 行 |
| Functional（方式 1） | ~45 行 |
| Functional（方式 2，带 @task） | ~55 行 |

**Functional API 更简洁**。

### 可读性

**StateGraph**：图结构一目了然

```python
graph.add_edge(START, "llm")
graph.add_conditional_edges("llm", should_continue, {"tools": "tools", "end": END})
graph.add_edge("tools", "llm")
```

**Functional**：逻辑流程一目了然

```python
while True:
    response = call_llm(messages)
    if not response.tool_calls:
        return response.content
    tool_results = execute_tools(response.tool_calls)
    messages.extend(tool_results)
```

### 灵活性

**StateGraph**：如果将来要添加复杂的路由（比如根据工具名路由到不同的节点），StateGraph 更容易扩展。

**Functional**：如果逻辑始终是"LLM → 工具 → LLM"的循环，Functional API 已经够用。

## 添加流式输出

Functional API 也可以流式输出：

```python
@task
def call_llm(messages: list, writer):
    writer.write("🤔 正在思考...")
    response = llm_with_tools.invoke(messages)
    return response

@task
def execute_tools(tool_calls, writer):
    writer.write(f"🔧 正在调用 {len(tool_calls)} 个工具...")
    results = []
    for tool_call in tool_calls:
        tool_name = tool_call["name"]
        writer.write(f"  - 调用 {tool_name}")
        
        tool_func = tool_map[tool_name]
        result = tool_func.invoke(tool_call["args"])
        
        results.append(ToolMessage(content=result, tool_call_id=tool_call["id"]))
    return results

@entrypoint
def react_agent(messages: list) -> str:
    while True:
        response = call_llm(messages)
        messages.append(response)
        
        if not response.tool_calls:
            return response.content
        
        tool_results = execute_tools(response.tool_calls)
        messages.extend(tool_results)

# 流式执行
for event in react_agent.stream([{"role": "user", "content": "搜索：天气"}]):
    print(event)
```

## 小结

用 Functional API 实现 ReAct Agent：

**优点**：
- 代码更简洁（少 30% 的代码）
- 逻辑更直观（就是一个 while 循环）
- 不需要定义 State、graph

**缺点**：
- 图结构不如 StateGraph 清晰
- 如果将来要扩展复杂路由，可能需要重构

**结论**：对于简单的 ReAct Agent，Functional API 是更好的选择。如果将来要添加复杂的多工具路由、并行调用等，再迁移到 StateGraph。

Functional API 适合简单到中等复杂度的工作流，代码更简洁、直观。对于复杂的动态路由，StateGraph API 更灵活。
