# 8.6 实战：研究员 + 写手 + 审核员协作系统

这一节我们综合前面学的内容，构建一个完整的多 Agent 协作系统——**内容生产流水线**。

---

## 系统架构

```
用户输入主题
    ↓
[Supervisor] → 分配任务
    ↓
[Researcher] → 搜索资料、收集数据
    ↓
[Writer] → 撰写文章
    ↓
[Reviewer] → 审核质量、提出修改意见
    ↓
[Supervisor] → 判断是否需要修改
    ├─ 需要修改 → 回到 Writer
    └─ 不需要 → 输出最终结果
```

---

## 完整代码

```python
from dotenv import load_dotenv
from langgraph.graph import StateGraph, START, END
from langchain_openai import ChatOpenAI
from langchain_core.messages import BaseMessage, HumanMessage, AIMessage, SystemMessage
from typing import TypedDict, Annotated
import operator

load_dotenv()

# ========== State ==========
class ContentPipelineState(TypedDict):
    messages: Annotated[list[BaseMessage], operator.add]
    topic: str
    research_data: str
    draft: str
    review_result: str
    revision_count: int

# ========== Researcher Agent ==========
researcher_llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.7)

def researcher_node(state: ContentPipelineState):
    """研究员：搜索资料，收集数据"""
    print("\n" + "="*50)
    print("📚 研究员：搜索资料中...")
    
    response = researcher_llm.invoke([
        SystemMessage(content="""你是一位研究专家。根据用户提供的主题，提供详细的研究资料。
包括：
1. 最新的研究进展
2. 核心概念和原理
3. 实际应用案例
4. 未来发展趋势
"""),
        HumanMessage(content=f"研究主题：{state['topic']}")
    ])
    
    print(f"✅ 研究完成，收集了 {len(response.content)} 字的资料")
    return {"research_data": response.content, "messages": [response]}

# ========== Writer Agent ==========
writer_llm = ChatOpenAI(model="gpt-4o", temperature=0.8)

def writer_node(state: ContentPipelineState):
    """写手：撰写文章"""
    print("\n" + "="*50)
    print("✍️ 写手：撰写文章中...")
    
    messages = [
        SystemMessage(content="""你是一位专业写作专家。根据研究资料和审核意见，撰写高质量的文章。

要求：
- 结构清晰，逻辑连贯
- 语言流畅，通俗易懂
- 有深度，有见解
"""),
    ]
    
    # 添加研究资料
    messages.append(HumanMessage(content=f"研究资料：\n{state['research_data']}"))
    
    # 如果有审核意见，加入上下文
    if state.get("review_result"):
        messages.append(HumanMessage(content=f"审核意见：\n{state['review_result']}\n\n请根据意见修改文章。"))
    
    # 添加主题
    messages.append(HumanMessage(content=f"文章主题：{state['topic']}"))
    
    response = writer_llm.invoke(messages)
    
    print(f"✅ 文章撰写完成，{len(response.content)} 字")
    return {"draft": response.content, "messages": [response]}

# ========== Reviewer Agent ==========
reviewer_llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.3)

def reviewer_node(state: ContentPipelineState):
    """审核员：审核文章质量"""
    print("\n" + "="*50)
    print("🔍 审核员：审核文章中...")
    
    response = reviewer_llm.invoke([
        SystemMessage(content="""你是一位严格的审核专家。审核文章的质量，从以下维度评估：

1. 事实准确性
2. 逻辑连贯性
3. 语言表达
4. 深度和见解

给出评分（1-10 分）和具体修改建议。

如果评分 >= 8 分，回复 "APPROVED"。
如果评分 < 8 分，列出需要修改的地方。
"""),
        HumanMessage(content=f"请审核以下文章：\n\n{state['draft']}")
    ])
    
    print(f"✅ 审核完成")
    print(f"审核结果：{response.content[:200]}...")
    
    return {"review_result": response.content, "messages": [response], "revision_count": state.get("revision_count", 0) + 1}

# ========== Supervisor Agent ==========
supervisor_llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

def supervisor_node(state: ContentPipelineState):
    """Supervisor：判断是否完成"""
    print("\n" + "="*50)
    print("👔 Supervisor：评估中...")
    
    revision_count = state.get("revision_count", 0)
    
    # 如果已经修改了 3 次，强制通过
    if revision_count >= 3:
        print("⚠️ 已修改 3 次，强制通过")
        return {"messages": [AIMessage(content="✅ 文章已通过审核")]}
    
    # 检查审核结果
    review_result = state.get("review_result", "")
    
    if "APPROVED" in review_result.upper():
        print("✅ 文章已通过审核")
        return {"messages": [AIMessage(content="✅ 文章已通过审核，任务完成")]}
    else:
        print("🔄 文章需要修改，返回写手")
        return {"messages": [AIMessage(content="🔄 文章需要修改")]}

# ========== 路由逻辑 ==========
def route_after_supervisor(state: ContentPipelineState):
    """Supervisor 之后的路由"""
    review_result = state.get("review_result", "")
    revision_count = state.get("revision_count", 0)
    
    if revision_count >= 3:
        return "end"
    elif "APPROVED" in review_result.upper():
        return "end"
    else:
        return "writer"

def route_after_reviewer(state: ContentPipelineState):
    """审核员之后的路由"""
    return "supervisor"

# ========== 构建图 ==========
graph = StateGraph(ContentPipelineState)

# 添加节点
graph.add_node("researcher", researcher_node)
graph.add_node("writer", writer_node)
graph.add_node("reviewer", reviewer_node)
graph.add_node("supervisor", supervisor_node)

# 添加边
graph.add_edge(START, "researcher")
graph.add_edge("researcher", "writer")
graph.add_edge("writer", "reviewer")
graph.add_edge("reviewer", "supervisor")

# Supervisor 之后的条件路由
graph.add_conditional_edges(
    "supervisor",
    route_after_supervisor,
    {
        "writer": "writer",
        "end": END
    }
)

# 编译
app = graph.compile()

# ========== 测试 ==========
if __name__ == "__main__":
    print("\n🚀 开始内容生产流水线\n")
    
    result = app.invoke({
        "topic": "人工智能在医疗领域的应用",
        "messages": [],
        "research_data": "",
        "draft": "",
        "review_result": "",
        "revision_count": 0
    })
    
    print("\n" + "="*50)
    print("📄 最终文章：")
    print("="*50)
    print(result["draft"])
    print("\n" + "="*50)
    print(f"修改次数：{result['revision_count']}")
    print(f"总消息数：{len(result['messages'])}")
```

## 扩展方向

1. **并行研究**：多个 Researcher 同时搜索不同方面的资料
2. **多轮审核**：Reviewer 也可以有多轮反馈
3. **专业审核员**：不同类型的文章交给不同的审核员
4. **持久化**：保存中间结果，支持暂停/恢复
5. **流式输出**：实时显示每个 Agent 的进度

---

## 小结

通过这个实战项目，我们综合运用了：

1. **Supervisor 模式**：中心化的任务分配
2. **多 Agent 协作**：研究员、写手、审核员各司其职
3. **条件路由**：根据审核结果决定是否需要修改
4. **循环**：写手和审核员之间可以循环，直到质量达标

这就是多 Agent 系统的魅力——通过分工协作，让每个 Agent 专注做好自己擅长的事。
